// @vitest-environment jsdom
// TEMPORARY probe for risk review (deleted after run, not committed).
import { afterEach, expect, test } from 'vitest';
import { cleanup, fireEvent, render, screen } from '@testing-library/react';
import userEvent from '@testing-library/user-event';
import App from './App';
afterEach(cleanup);

test('probe A: time inputs reachable by their visible label', () => {
  render(<App />);
  expect(() => screen.getByLabelText('Start time')).not.toThrow();
});

test('probe B: invalid (cleared) start time shows a visible message', () => {
  render(<App />);
  fireEvent.change(screen.getByDisplayValue('11:00'), { target: { value: '' } });
  expect(screen.queryByText('Invalid time')).not.toBeNull();
});

test('probe C: Check with cleared start time — what does the user see?', async () => {
  render(<App />);
  fireEvent.change(screen.getByDisplayValue('11:00'), { target: { value: '' } });
  await userEvent.setup().click(screen.getByRole('button', { name: 'Check' }));
  console.log('PROBE C banner:', screen.getByText(/Please select|valid/).textContent);
});
